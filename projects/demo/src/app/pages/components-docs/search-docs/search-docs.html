Search

Site search and command palette with section grouping, grapheme-cluster-safe substring highlighting, recent history persistence, and keyboard roving navigation.

Overview

The Search component delivers a fast, accessible command palette experience designed to sit inside an <x-liquid-menu-popup> or standalone container. Arrow keys navigate through grouped results while keeping focus inside the search field, and matches are highlighted without breaking Unicode grapheme clusters.

Contexts & Overlays

Search is designed to pair natively with LiquidMenuPopup to provide a floating global command palette anchored to a header trigger.

Modal Command Palette Anchored popup dialog with lazy content projection and auto-focus
Embedded Sidebar Filter Fixed in-flow catalog navigation

Setup & Usage

import { Search, type SearchItem } from '@app/shared/components/search/search';
import { LiquidMenuPopup } from '@app/shared/components/liquid-menu-popup/liquid-menu-popup';
import { LiquidMenuContent } from '@app/shared/components/liquid-menu-popup/liquid-menu-content.directive';
import { LiquidMenuTrigger } from '@app/shared/components/liquid-menu-popup/liquid-menu-trigger.directive';

@Component({
  imports: [Search, LiquidMenuPopup, LiquidMenuContent, LiquidMenuTrigger],
  template: `
    <button type="button" [liquidMenuTriggerFor]="searchMenu">Search</button>

    <x-liquid-menu-popup #searchMenu="liquidMenu" role="dialog" title="Search">
      <ng-template liquidMenuContent>
        <app-search [items]="searchItems" />
      </ng-template>
    </x-liquid-menu-popup>
  `,
})
export class HeaderNavigation {
  protected readonly searchItems: SearchItem[] = [
    { title: 'Cluster Overview', path: '/monitoring', section: 'Analytics', icon: 'ionGlobeOutline' },
  ];
}
